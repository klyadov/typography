
--  Создание базы данных (CREATE DATABASE)
CREATE DATABASE typography;
USE typography_db;

-- Создание таблиц и связей (CREATE TABLE)
CREATE TABLE workshop (
    id_workshop INT PRIMARY KEY,
    title_workshop VARCHAR(150) NOT NULL
);

CREATE TABLE employee (
    id_employee INT PRIMARY KEY,
    full_name VARCHAR(150) NOT NULL,
    position VARCHAR(100) NOT NULL,
    id_workshop INT NOT NULL,
    FOREIGN KEY (id_workshop) REFERENCES workshop(id_workshop)
);

CREATE TABLE customer (
    id_customer INT PRIMARY KEY,
    name_customer VARCHAR(150) NOT NULL,
    inn_customer VARCHAR(12) NOT NULL,
    address VARCHAR(255) NOT NULL,
    phone VARCHAR(20) NOT NULL
);

CREATE TABLE material (
    id_material INT PRIMARY KEY,
    title_material VARCHAR(150) NOT NULL
);

CREATE TABLE equipment (
    id_equipment INT PRIMARY KEY,
    title_equipment VARCHAR(100) NOT NULL
);

CREATE TABLE tech_card (
    id_tech_card INT PRIMARY KEY,
    product_type VARCHAR(100) NOT NULL,
    format_size VARCHAR(50) NOT NULL,
    id_material INT NOT NULL,
    FOREIGN KEY (id_material) REFERENCES material(id_material)
);

CREATE TABLE print_order (
    id_order INT PRIMARY KEY,
    date_created DATETIME NOT NULL,
    date_deadline DATETIME NOT NULL,
    id_customer INT NOT NULL,
    id_employee INT NOT NULL,
    FOREIGN KEY (id_customer) REFERENCES customer(id_customer),
    FOREIGN KEY (id_employee) REFERENCES employee(id_employee)
);

CREATE TABLE order_detail (
    id_detail INT PRIMARY KEY,
    id_order INT NOT NULL,
    id_tech_card INT NOT NULL,
    quantity INT NOT NULL,
    price DECIMAL(10,2) NOT NULL,
    FOREIGN KEY (id_order) REFERENCES print_order(id_order),
    FOREIGN KEY (id_tech_card) REFERENCES tech_card(id_tech_card)
);

CREATE TABLE production_task (
    id_task INT PRIMARY KEY,
    id_detail INT NOT NULL,
    id_equipment INT NOT NULL,
    FOREIGN KEY (id_detail) REFERENCES order_detail(id_detail),
    FOREIGN KEY (id_equipment) REFERENCES equipment(id_equipment)
);

-- Неявное соединение таблиц с фильтром (WHERE) и сортировкой (ORDER BY)
SELECT po.id_order, c.name_customer, tc.product_type, od.price
FROM print_order po, customer c, order_detail od, tech_card tc
WHERE po.id_customer = c.id_customer
  AND od.id_order = po.id_order
  AND od.id_tech_card = tc.id_tech_card
  AND od.price >= 45000
ORDER BY od.price DESC;

-- Явное соединение таблиц (JOIN)
SELECT po.id_order, c.name_customer, e.full_name, tc.product_type, od.quantity, od.price
FROM print_order po
JOIN customer c ON po.id_customer = c.id_customer
JOIN employee e ON po.id_employee = e.id_employee
JOIN order_detail od ON po.id_order = od.id_order
JOIN tech_card tc ON od.id_tech_card = tc.id_tech_card;

--. Агрегирующие операции (SUM, AVG, COUNT, MAX, MIN)
SELECT 
    COUNT(*) AS total_items,
    SUM(price) AS total_sum,
    AVG(price) AS avg_price,
    MIN(price) AS min_price,
    MAX(price) AS max_price
FROM order_detail;

-- Группировка GROUP BY и фильтрация HAVING
SELECT id_order, COUNT(*) AS count_items, SUM(price) AS total_sum
FROM order_detail
GROUP BY id_order
HAVING total_sum > 100000;

-- Вложенные подзапросы (IN, ALL)
-- Подзапрос с IN: клиенты с крупными тиражами (от 5 000 экз.)
SELECT id_customer, name_customer, phone
FROM customer
WHERE id_customer IN (
    SELECT po.id_customer
    FROM print_order po
    JOIN order_detail od ON po.id_order = od.id_order
    WHERE od.quantity >= 5000
);

-- Подзапрос с ALL: позиции дороже каждой из позиций заказа № 1002
SELECT id_detail, id_order, price
FROM order_detail
WHERE price > ALL (
    SELECT price
    FROM order_detail
    WHERE id_order = 1002
);

-- Функции управления потоком (IF, CASE)
-- Функция IF: деление тиражей на оптовые и стандартные
SELECT 
    id_detail,
    quantity,
    price,
    IF(quantity >= 3000, 'Крупный оптовый тираж', 'Стандартный тираж') AS order_scale
FROM order_detail;

-- Оператор CASE: категоризация маржинальности заказа
SELECT 
    id_order,
    price,
    CASE 
        WHEN price >= 200000.00 THEN 'Категория А: VIP-заказ'
        WHEN price >= 100000.00 THEN 'Категория B: Крупный заказ'
        ELSE 'Категория C: Базовый заказ'
    END AS price_category
FROM order_detail;

-- Представление (CREATE VIEW) и временное представление (WITH / CTE)
-- Постоянное представление на популярный запрос сводки заказов
CREATE VIEW vw_popular_orders_summary AS
SELECT 
    po.id_order,
    c.name_customer,
    tc.product_type,
    tc.format_size,
    od.quantity,
    od.price,
    po.date_deadline
FROM print_order po
JOIN customer c ON po.id_customer = c.id_customer
JOIN order_detail od ON po.id_order = od.id_order
JOIN tech_card tc ON od.id_tech_card = tc.id_tech_card;

SELECT * FROM vw_popular_orders_summary;

-- Временное представление с оператором WITH
WITH client_total_revenue AS (
    SELECT 
        c.id_customer,
        c.name_customer,
        SUM(od.price) AS total_spent
    FROM customer c
    JOIN print_order po ON c.id_customer = po.id_customer
    JOIN order_detail od ON po.id_order = od.id_order
    GROUP BY c.id_customer, c.name_customer
)
SELECT name_customer, total_spent
FROM client_total_revenue
WHERE total_spent > 100000.00;
